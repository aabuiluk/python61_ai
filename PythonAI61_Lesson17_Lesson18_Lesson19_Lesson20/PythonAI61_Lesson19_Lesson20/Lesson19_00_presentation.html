На попередніх парах MLP уже навчалась, але тепер важливо зрозуміти, чому одні hidden-активації пропускають градієнт краще за інші, коли потрібен Sigmoid або Softmax на виході і як clip_grad_norm_ рятує від вибуху кроку.

"""Runnable demo for pair 19: Зникаючі градієнти в стеку sigmoid, hooks на норми, ініціалізація Glorot."""

from __future__ import annotations

try:
    import torch
    import torch.nn as nn

    def grad_norms(init: str) -> list[float]:
        layers = [nn.Linear(8, 8) for _ in range(6)]
        if init == 'xavier':
            for layer in layers:
                nn.init.xavier_uniform_(layer.weight)
        model = nn.Sequential(*[m for layer in layers for m in (layer, nn.Sigmoid())])
        norms: list[float] = []
        hooks = [layers[i].weight.register_hook(lambda g, i=i: norms.append(float(g.norm()))) for i in range(6)]
        x = torch.randn(16, 8)
        model(x).sum().backward()
        for h in hooks:
            h.remove()
        return norms

    plain = grad_norms('plain')
    xavier = grad_norms('xavier')
    print('plain', [round(v, 4) for v in plain])
    print('xavier', [round(v, 4) for v in xavier])
    assert len(plain) == 6
except ModuleNotFoundError as exc:
    print('skip_missing_dep', getattr(exc, 'name', None) or exc)
