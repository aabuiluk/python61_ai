"""Runnable demo for pair 20: UCI Heart Disease: поріг під recall і model card «це не діагноз»."""

from __future__ import annotations

try:
    import numpy as np
    from sklearn.metrics import confusion_matrix, recall_score

    y = np.array([0, 0, 0, 1, 1, 1, 1, 1])
    p = np.array([0.10, 0.20, 0.45, 0.40, 0.55, 0.70, 0.80, 0.30])

    def at(thr: float) -> tuple[float, list[list[int]]]:
        yhat = (p >= thr).astype(int)
        return float(recall_score(y, yhat)), confusion_matrix(y, yhat).tolist()

    thr = 0.5
    while at(thr)[0] < 0.9 and thr > 0.05:
        thr -= 0.05
    rec, cm = at(thr)
    print('threshold', round(thr, 2), 'recall', rec, 'cm', cm)
    assert rec >= 0.9
    card = {
        'intended_use': 'classroom threshold demo on UCI Heart Disease',
        'out_of_scope': 'not a diagnosis, not a clinical device',
        'fallback': 'WDBC analog only if Heart table is unavailable',
    }
    print(card)
except ModuleNotFoundError as exc:
